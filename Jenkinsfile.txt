
pipeline {
    agent any
    parameters {
    choice(
        name: 'RELAY_SCENARIO',
        choices: [
            'ALL',
            'NORMAL',
            'MISMATCH',
            'COMMUNICATION_FAILURE'
        ],
        description: 'Select relay test scenario'
    )
}

    tools {
        jdk 'JDK17'
        maven 'Maven'
    }

    stages {
        stage('Checkout') {
            steps {
                echo 'Smart Relay Monitoring System'
            }
        }

        stage('Build and Test') {
            steps {
                bat 'mvn clean test'
            }
        }

        stage('Package') {
            steps {
                bat 'mvn package'
            }
        }
    }

    post {
        success {
            echo 'Build and tests completed successfully!'
        }
        failure {
            echo 'Build failed. Please check the console output.'
        }
    }
}
