
pipeline {
    agent any

    options {
        skipDefaultCheckout(true)
    }

    stages {

        stage('Checkout') {
            steps {
                echo 'Checking out project from GitHub'
                checkout scm
            }
        }

        stage('Build and Test') {
            steps {
                echo 'Building project and running JUnit tests'
                bat 'mvn clean package'
            }
        }
        stage('Run Selected Relay Scenario') {
    steps {
        bat 'java -jar target\\SmartRelayMonitoringSystem-1.0-SNAPSHOT.jar "%RELAY_SCENARIO%"'
    }
}

        stage('Generate HTML Report') {
            steps {
                echo 'Generating HTML test report'
                bat 'mvn surefire-report:report-only'
            }
        }

        stage('Publish JUnit Results') {
            steps {
                junit 'target/surefire-reports/*.xml'
            }
        }

        stage('Publish HTML Report') {
            steps {
                publishHTML(target: [
                    reportName: 'JUnit Test Report',
                    reportDir: 'target/reports',
                    reportFiles: 'surefire.html',
                    keepAll: true,
                    alwaysLinkToLastBuild: true,
                    allowMissing: false
                ])
            }
        }
    }

    post {
        success {
            echo 'Build and tests completed successfully!'
            echo 'HTML Test Report published successfully!'
        }

        failure {
            echo 'Build failed. Please check the console output.'
        }

        always {
            echo 'Jenkins pipeline execution completed.'
        }
    }
}
