
pipeline {
    agent any
    parameters {
    choice(
        name: 'RELAY_SCENARIO',
        choices: [
            'ALL',
            'NORMAL',
            'MISMATCH',
            'COMMUNICATION_FAILURE'
        ],
        description: 'Select relay test scenario'
    )
}

    tools {
        jdk 'JDK17'
        maven 'Maven'
    }

    stages {
        stage('Checkout') {
            steps {
                echo 'Smart Relay Monitoring System'
            }
        }

        stage('Build and Test') {
            steps {
                bat 'mvn clean test'
            }
        }
        stage('Generate HTML Report') {
    steps {
        bat 'mvn -B -Dmaven.repo.local="%WORKSPACE%\\.m2\\repository" surefire-report:report'
    }
}
stage('Publish HTML Report') {
    steps {
        publishHTML(target: [
            reportDir: 'target/site',
            reportFiles: 'surefire-report.html',
            reportName: 'JUnit Test Report',
            keepAll: true,
            alwaysLinkToLastBuild: true,
            allowMissing: false
        ])
    }
}

        stage('Package') {
            steps {
                bat 'mvn package'
            }
        }
    }

    post {
        success {
            echo 'Build and tests completed successfully!'
        }
        failure {
            echo 'Build failed. Please check the console output.'
        }
    }
}
